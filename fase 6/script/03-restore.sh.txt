#!/usr/bin/env bash
set -Eeuo pipefail

PROJECT="final-equipeN"
POOL="pool-final"

TYPE=""; RESOURCE=""; SNAP=""; DRY_RUN=0; YES=0
while [ $# -gt 0 ]; do
  case "$1" in
    --type) TYPE="$2"; shift 2 ;;
    --resource) RESOURCE="$2"; shift 2 ;;
    --snapshot) SNAP="$2"; shift 2 ;;
    --dry-run) DRY_RUN=1; shift ;;
    --yes) YES=1; shift ;;
    --help) echo "Uso: $0 --type instance|volume --resource NOME --snapshot snap-XXXX [--dry-run] [--yes]"; echo "Ex: $0 --type instance --resource app --snapshot snap-20261122-180000"; echo "Ex: $0 --type volume --resource vol-db --snapshot snap-20261122-180000"; exit 0 ;;
    *) echo "[ERRO] Arg desconhecido: $1" >&2; exit 2 ;;
  esac
done

# Compatibilidade com forma antiga: ./03-restore.sh app snap-xxx
if [ -z "$TYPE" ] && [ -n "$RESOURCE" ] && [ -z "$SNAP" ]; then :; fi

[ -n "$TYPE" ] || { echo "[ERRO] Informe --type instance|volume" >&2; exit 2; }
[ -n "$RESOURCE" ] || { echo "[ERRO] Informe --resource NOME" >&2; exit 2; }
[ -n "$SNAP" ] || { echo "[ERRO] Informe --snapshot snap-XXXX" >&2; exit 2; }
[[ "$TYPE" == "instance" || "$TYPE" == "volume" ]] || { echo "[ERRO] --type deve ser instance ou volume" >&2; exit 2; }

command -v lxc >/dev/null || { echo "[ERRO] lxc nao encontrado" >&2; exit 1; }

run() { if [ "$DRY_RUN" -eq 1 ]; then echo "[DRY-RUN] $*"; else "$@"; fi; }

# Validacoes
if [ "$TYPE" = "instance" ]; then
  lxc info --project "$PROJECT" "$RESOURCE" >/dev/null 2>&1 || { echo "[ERRO] Instancia $RESOURCE nao existe no projeto $PROJECT" >&2; exit 1; }
  lxc snapshot list --project "$PROJECT" "$RESOURCE" --format csv | grep -qx "$SNAP" || { echo "[ERRO] Snapshot $SNAP nao existe em $RESOURCE" >&2; exit 1; }
else
  lxc storage volume list "$POOL" --format csv | grep -q "^${RESOURCE}," || { echo "[ERRO] Volume $RESOURCE nao existe no pool $POOL" >&2; exit 1; }
  lxc storage volume snapshot list "$POOL" "$RESOURCE" --format csv | grep -qx "$SNAP" || { echo "[ERRO] Snapshot $SNAP nao existe no volume $RESOURCE" >&2; exit 1; }
fi

echo "[AVISO] Isso vai SUBSTITUIR os dados atuais de $TYPE $RESOURCE pelo snapshot $SNAP."
if [ "$YES" -eq 0 ] && [ "$DRY_RUN" -eq 0 ]; then
  read -r -p "Digite CONFIRMAR para continuar: " ans
  [ "$ans" = "CONFIRMAR" ] || { echo "[INFO] Cancelado pelo usuario"; exit 0; }
fi

if [ "$TYPE" = "instance" ]; then
  echo "[INFO] Parando $RESOURCE..."
  run lxc stop --project "$PROJECT" "$RESOURCE" || { echo "[ERRO] Falha ao parar $RESOURCE" >&2; exit 1; }
  echo "[INFO] Restaurando $RESOURCE -> $SNAP"
  run lxc restore --project "$PROJECT" "$RESOURCE" "$SNAP" || { echo "[ERRO] Falha no restore. Tente iniciar manualmente: lxc start --project $PROJECT $RESOURCE" >&2; exit 1; }
  echo "[INFO] Iniciando $RESOURCE..."
  run lxc start --project "$PROJECT" "$RESOURCE" || { echo "[ERRO] Falha ao iniciar $RESOURCE apos restore" >&2; exit 1; }
else
  # Volume: precisa descobrir quem usa e parar
  echo "[INFO] Volume $RESOURCE - verifique qual instancia usa este volume antes de restaurar."
  echo "[INFO] Exemplo: vol-db -> db, vol-uploads -> app, dev-vol-db -> dev-db, dev-vol-uploads -> dev-app"
  # Mapa simples
  case "$RESOURCE" in
    vol-db) OWNER="db" ;; vol-uploads) OWNER="app" ;;
    dev-vol-db) OWNER="dev-db" ;; dev-vol-uploads) OWNER="dev-app" ;;
    *) OWNER="" ;;
  esac
  if [ -n "$OWNER" ]; then
    echo "[INFO] Parando instancia dona $OWNER..."
    run lxc stop --project "$PROJECT" "$OWNER" || { echo "[ERRO] Falha ao parar $OWNER" >&2; exit 1; }
  fi
  echo "[INFO] Restaurando volume $RESOURCE -> $SNAP"
  run lxc storage volume restore "$POOL" "$RESOURCE" "$SNAP" || { echo "[ERRO] Falha restore volume" >&2; exit 1; }
  if [ -n "$OWNER" ]; then
    echo "[INFO] Iniciando $OWNER..."
    run lxc start --project "$PROJECT" "$OWNER" || { echo "[ERRO] Falha ao iniciar $OWNER" >&2; exit 1; }
  fi
fi

echo "[OK] Restore concluido: $TYPE $RESOURCE -> $SNAP"
echo "Verifique: lxc info --project $PROJECT $RESOURCE  (ou) lxc storage volume list $POOL"
