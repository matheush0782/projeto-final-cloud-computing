#!/usr/bin/env bash
set -Eeuo pipefail

PROJECT="final-equipeN"
POOL="pool-final"
TS="$(date +%Y%m%d-%H%M%S)"
SNAP="snap-${TS}"

INSTANCES_PROD=("proxy" "app" "db")
INSTANCES_DEV=("dev-proxy" "dev-app" "dev-db")
VOLUMES_PROD=("vol-db" "vol-uploads")
VOLUMES_DEV=("dev-vol-db" "dev-vol-uploads")

DRY_RUN=0
CONSISTENT=0
for arg in "$@"; do
  case "$arg" in
    --dry-run) DRY_RUN=1 ;;
    --consistent) CONSISTENT=1 ;;
    --help) echo "Uso: $0 [--dry-run] [--consistent]"; echo "  --consistent: para app/db antes de snapshottar p/ consistencia e religa depois"; exit 0 ;;
    *) echo "[ERRO] Argumento desconhecido: $arg" >&2; exit 2 ;;
  esac
done

log() { echo "[INFO] $*"; }
ok() { echo "[OK] $*"; }
err() { echo "[ERRO] $*" >&2; }

command -v lxc >/dev/null || { err "CLI 'lxc' nao encontrada"; exit 1; }
if ! lxc project list --format csv | grep -qx "$PROJECT"; then err "Projeto $PROJECT nao existe"; exit 1; fi

run() {
  if [ "$DRY_RUN" -eq 1 ]; then echo "[DRY-RUN] $*"; else "$@"; fi
}

# Consistencia: se --consistent, para app/db antes. Proxy nao precisa parar.
# Ordem: app primeiro, depois db. Volta na ordem inversa.
if [ "$CONSISTENT" -eq 1 ] && [ "$DRY_RUN" -eq 0 ]; then
  log "Modo consistente: parando app e db (PROD e DEV) para snapshot consistente"
  for inst in app dev-app db dev-db; do
    lxc stop --project "$PROJECT" "$inst" || err "Falha ao parar $inst (continuando)"
  done
fi

# Garante que religa mesmo se falhar no meio
cleanup() {
  if [ "$CONSISTENT" -eq 1 ] && [ "$DRY_RUN" -eq 0 ]; then
    log "Religando instancias paradas"
    for inst in db dev-db app dev-app; do
      lxc start --project "$PROJECT" "$inst" || err "Falha ao iniciar $inst - verifique manualmente"
    done
  fi
}
trap cleanup EXIT

FAILED=0
log "Criando snapshots de instancias - ponto: $SNAP"

for inst in "${INSTANCES_PROD[@]}" "${INSTANCES_DEV[@]}"; do
  if ! lxc info --project "$PROJECT" "$inst" >/dev/null 2>&1; then err "Instancia $inst nao existe, pulando"; FAILED=1; continue; fi
  log "Snapshot instancia $inst -> $SNAP"
  if run lxc snapshot --project "$PROJECT" "$inst" "$SNAP"; then ok "$inst"; else err "Falha snapshot $inst"; FAILED=1; fi
done

log "Criando snapshots de volumes - pool $POOL - ponto: $SNAP"
for vol in "${VOLUMES_PROD[@]}" "${VOLUMES_DEV[@]}"; do
  if ! lxc storage volume list "$POOL" --format csv | grep -q "^${vol},"; then err "Volume $vol nao existe, pulando"; FAILED=1; continue; fi
  log "Snapshot volume $vol -> $SNAP"
  # Sintaxe LXD: lxc storage volume snapshot <pool> <volume> <snapshot>
  if run lxc storage volume snapshot "$POOL" "$vol" "$SNAP"; then ok "$vol"; else err "Falha snapshot volume $vol"; FAILED=1; fi
done

if [ "$FAILED" -ne 0 ]; then err "Snapshot concluido com FALHAS parciais. Verifique acima. Ponto: $SNAP"; exit 1; fi
ok "Snapshots concluidos. Ponto de recuperacao: $SNAP"
echo "Instancias: proxy, app, db, dev-proxy, dev-app, dev-db"
echo "Volumes: vol-db, vol-uploads, dev-vol-db, dev-vol-uploads"
